/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public boolean isBalanced(TreeNode root) {
        return getHeight(root) != -1;
    }

    private int getHeight(TreeNode node) { //计算子树高度，不平衡返回-1
        if (node == null) return 0;   //递归出口，空树高为0

        int leftH = getHeight(node.left); //左高
        if (leftH == -1) return -1;
        
        int rightH = getHeight(node.right); //右高
        if (rightH == -1) return -1;

        if (Math.abs(leftH -rightH) > 1) { //高度差超过1，返回-1
            return -1;
        }
        return Math.max(leftH, rightH) + 1; //为左右子树高的那一个
    }
}
