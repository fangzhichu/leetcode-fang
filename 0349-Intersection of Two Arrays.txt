import java.util.HashSet;
import java.util.Set;

class Solution {
    public int[] intersection(int[] nums1, int[] nums2) {
        Set<Integer> set1 = new HashSet<>();
        Set<Integer> res = new HashSet<>();

        for (int x : nums1) {
            set1.add(x);
        }
        for (int x : nums2) {
            if (set1.contains(x)) {
                res.add(x);
            }
        }

        int[] ans = new int[res.size()];
        int idx = 0;
        for (int val : res) {
            ans[idx++] = val;
        }
        return ans;
    }
}
